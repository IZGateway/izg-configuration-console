import { Page, expect, test } from '@playwright/test';
import { loginToOkta } from '../helpers/oktaLogin';

let context;
let page;

test.beforeAll(async ({ browser }) => {
    context = await browser.newContext();
    page = await context.newPage();
    await loginToOkta(page, process.env.OKTA_USERNAME, process.env.OKTA_PASSWORD);
    await page.goto('/manageconnections'); // Update based on your test
    await page.waitForLoadState('networkidle'); // Ensure the page is fully loaded
});

test.afterAll(async () => {
  await logout(page)
  await page.close()
  await context.close()
})

test('Test', async () => {
  await page.pause(); 
});
